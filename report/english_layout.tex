% Both cover pages contain each label, so patch both occurrences explicitly.
\newcommand{\translateCoverLabel}[2]{%
  \patchcmd{\makeCover}{#1}{#2}{}
    {\PackageError{leaf-hierarchy}{Could not translate a cover label}{Check the upstream cover template.}}%
  \patchcmd{\makeCover}{#1}{#2}{}
    {\PackageError{leaf-hierarchy}{Could not translate the second cover label}{Check the upstream cover template.}}%
}
\translateCoverLabel{\bfseries Autor}{\bfseries Author}
\translateCoverLabel{\bfseries Director}{\bfseries Supervisor}
\translateCoverLabel{\bfseries Colaborador}{\bfseries Collaborator}
